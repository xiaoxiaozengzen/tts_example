cmake_minimum_required(VERSION 3.21)
project(tts_example)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_BINARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/lib)

find_package(ALSA REQUIRED)
message(STATUS "ALSA_INCLUDE_DIRS: ${ALSA_INCLUDE_DIRS}")
message(STATUS "ALSA_LIBRARIES: ${ALSA_LIBRARIES}")
message(STATUS "ALSA_DIR: ${ALSA_DIR}")

add_subdirectory(ekho_example)

add_executable(tts_example src/main.cpp)
